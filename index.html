// --- BOS - Break of Structure --- if(UseBOS) { // Bullish BOS = breaks last high, Bearish BOS = breaks last low bullishBOS = (price > lastBOSHigh && lastBOSHigh > 0); bearishBOS = (price < lastBOSLow && lastBOSLow > 0); double l1 = iLow(_Symbol, PERIOD_M15, 3); double h2 = iHigh(_Symbol, PERIOD_M15, 2); double l2 = iLow(_Symbol, PERIOD_M15, 2); double h3 = iHigh(_Symbol, PERIOD_M15, 1); double l3 = iLow(_Symbol, PERIOD_M15, 1); } } else { bullishFVG = true; bearishFVG = true; } // --- DERIV REAL MARKET FILTER --- bool derivSell = true; bool derivBuy = true; if(UseDerivFilter) { // Deriv real XAUUSD is same as broker - We use H4 alignment as Deriv truth // If Deriv real also SELL (H4 below), confirm derivSell = h4Sell && h1Sell; derivBuy = !h4Sell && !h1Sell; } int holdMinutes = (int)((TimeCurrent() - lastFlipTime)/60); bool canFlip = (holdMinutes >= MinHoldMinutes); // --- PERFECT EA LOGIC - Holds direction - Only flips on Sweep + BOS + FVG --- if(AllowFlipOnlyOnSweep && realSweep && canFlip) { // Flip SELL -> BUY only if: Sweep Low + Bullish BOS + Bullish FVG + Deriv Buy if(currentDirection == "SELL" && sweepLow && bullishBOS && bullishFVG && derivBuy) { currentDirection = "BUY"; } // Flip BUY -> SELL only if: Sweep High + Bearish BOS + Bearish FVG + Deriv Sell else if(currentDirection == "BUY" && sweepHigh && bearishBOS && bearishFVG && derivSell) { currentDirection = "SELL"; lastFlipTime = TimeCurrent(); Print("TRUTH ULTIMATE: 🔥 SWEEP HIGH + BOS BEAR + FVG BEAR + DERIV SELL - FLIP TO SELL - Held BUY ", holdMinutes, " min"); CloseAll(); trade.Sell(LotSize, _Symbol, bid, bid + StopLossPoints*_Point*10, bid - TakeProfitPoints*_Point*10, "TRUTH ULTIMATE SELL BOS FVG DERIV"); } } lastHighH1 = highH1; lastLowH1 = lowH1; // --- TRUTH WRITINGS - BUY HERE SELL HERE - On chart --- string info = "TRUTH ULTIMATE - DERIV + LIQ + BOS + FVG - FROM MQL5\n"; info += "----------------------------------------\n"; info += "H4: " + (h4Sell?"SELL":"BUY") + " H1: " + (h1Sell?"SELL":"BUY") + " M15: " + (m15Sell?"SELL":"BUY") + "\n"; info += "Liquidity: Range " + DoubleToString(rangeUSD, 2) + " USD - Sweep " + (realSweep?"YES FLIP ALLOWED":"NO HOLD") + "\n"; info += "BOS: " + (bullishBOS?"BULLISH BOS ":"") + (bearishBOS?"BEARISH BOS ":"") + " LastHigh " + DoubleToString(lastBOSHigh, 2) + "\n"; info += "FVG: " + (bullishFVG?"BULL FVG ":"") + (bearishFVG?"BEAR FVG ":"") + " M15\n"; info += "Deriv Filter: " + (derivSell?"DERIV SELL ":"") + (derivBuy?"DERIV BUY ":"") + "\n"; info += "Min Hold: " + IntegerToString(MinHoldMinutes) + " min - Last Flip: " + TimeToString(lastFlipTime) + "\n"; info += "Rule: SELL stays SELL until Sweep+BOS+FVG+Deriv\n"; info += "----------------------------------------\n"; info += "BUY HERE →\n"; info += "SELL HERE ←\n"; info += "TRUTH MODE - Focus Serious - No Fool Code"; Comment(info); // Auto entry if no position - Must match current direction + Deriv + BOS + FVG if(PositionsTotal() == 0) { if(currentDirection == "SELL" && h4Sell && h1Sell && bearishBOS && bearishFVG && derivSell) trade.Sell(LotSize, _Symbol, bid, bid + StopLossPoints*_Point*10, bid - TakeProfitPoints*_Point*10, "TRUTH ULTIMATE SELL - BOS FVG DERIV"); else if(currentDirection == "BUY" && !h4Sell && !h1Sell && bullishBOS && bullishFVG && derivBuy) trade.Buy(LotSize, _Symbol, ask, ask - StopLossPoints*_Point*10, ask + TakeProfitPoints*_Point*10, "TRUTH ULTIMATE BUY - BOS FVG DERIV"); } } //+------------------------------------------------------------------+ void CloseAll() { for(int i=PositionsTotal()-1; i>=0; i--) { string sym = PositionGetSymbol(i); if(sym == _Symbol) trade.PositionClose(_Symbol); } } double GetMA(ENUM_TIMEFRAMES tf, int period) { int h = iMA(_Symbol, tf, period, 0, MODE_EMA, PRICE_CLOSE); double buf[]; ArraySetAsSeries(buf, true); CopyBuffer(h, 0, 0, 1, buf); return buf[0]; } //+------------------------------------------------------------------+
